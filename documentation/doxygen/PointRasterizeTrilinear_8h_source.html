35/// @brief Traits for staggered (MAC) trilinear rasterization. This method
36/// only works on scalar or Vec3 attribute types and only ever returns a Vec3
37/// tree. It is typically used for velocity rasterization in fluid
38/// simulations to avoid the pressure-velocity decoupling problem that occurs
39/// on collocated (cell-centered) grids. The resulting Vec3 tree effectively
40/// represents the velocity values on the cell faces rather than the cell
41/// centers.
42template <typename ValueT, bool Staggered = true>
43struct TrilinearTraits
44{
45private:
46 using FltT = typename types_internal::flt_t<sizeof(typename ValueTraits<ValueT>::ElementType)*CHAR_BIT>::type;
47 static_assert(ValueTraits<ValueT>::IsScalar ||
48 (ValueTraits<ValueT>::IsVec && ValueTraits<ValueT>::Size == 3),
49 "Source attribute type must be scalar or Vec3 for staggered rasterization.");
50public:
51 /// @brief Resulting rasterized Tree ValueType (always Vec3)
52 using ResultT = math::Vec3<FltT>;
53 /// @brief Resulting rasterized TreeType
54 template <typename PointDataTreeT>
55 using TreeT = typename PointDataTreeT::template ValueConverter<ResultT>::Type;
56};
57
58/// @brief Traits for collocated, cell-centered rasterization. Work for any
59/// scalar, vector or matrix attribute. Note that integer values are
60/// interpolated at their respective floating point precision and returned
61/// as trees at that precision.
62template <typename ValueT>
63struct TrilinearTraits<ValueT, false>
64{
65private:
66 using FltT = typename types_internal::flt_t<sizeof(typename ValueTraits<ValueT>::ElementType)*CHAR_BIT>::type;
67public:
68 /// @brief Resulting rasterized Tree ValueType
69 using ResultT = typename ConvertElementType<ValueT, FltT>::Type;
70 /// @brief Resulting rasterized TreeType
71 template <typename PointDataTreeT>
72 using TreeT = typename PointDataTreeT::template ValueConverter<ResultT>::Type;
73};

75/// @brief Perform weighted trilinear rasterization of all points within a
76/// voxel. This method takes and returns a tree i.e. ignores grid
77/// transformations.
78/// @details Accumulates values and weights according to a simple 0-1-0 weighted
79/// hat function. This algorithm is an exact inverse of a trilinear
80/// interpolation and thus a key method used in PIC/FLIP style simulations.
81/// Returns a tree of the same precision as the input source attribute, but
82/// may be of a different math type depending on the value of the Staggered
83/// template attribute. If Staggered is true, this method produces values at
84/// each voxels negative faces, causing scalar attributes to produce
85/// math::Vec3<ValueT> tree types. The result Tree type is equal to:
86/// TrilinearTraits<ValueT, Staggered>::template TreeT<PointDataTreeT>
87/// @tparam Staggered whether to perform a staggered or collocated rasterization
88/// @tparam ValueT the value type of the point attribute to rasterize
89/// @param points the point tree to be rasterized
90/// @param attribute the name of the attribute to rasterize. Must be a scalar
91/// or Vec3 attribute for Staggered rasterization. Otherwise, can be any
92/// scalar, Vector or Matrix type. Integer values are interpolated at their
93/// float bitwidth precision and returned as a tree at that precision e.g:
94/// int32 -> float, FloatTree.
95/// @param filter an optional point filter to use
96template <bool Staggered,
97 typename ValueT,
98 typename FilterT = NullFilter,
99 typename PointDataTreeT = PointDataTree>
100inline auto
101rasterizeTrilinear(const PointDataTreeT& points,
102 const std::string& attribute,
103 const FilterT& filter = NullFilter());
