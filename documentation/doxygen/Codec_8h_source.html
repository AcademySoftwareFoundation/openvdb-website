41/// @brief Controls which operations a codec exposes.
42///
43/// Passed to codec registration to restrict how the codec may be used at
44/// runtime. A @c ReadOnly codec can still be used to read existing files but
45/// will not be selected as the write codec for new grids.
46enum class CodecMode {
47 /// Both @c readTopology()/@c readBuffers() and
48 /// @c writeTopology()/@c writeBuffers() are enabled. This is the default.
49 ReadWrite,
50 /// Only @c readTopology() and @c readBuffers() are enabled; the codec
51 /// will not be offered as a write target.
52 ReadOnly
53};
54
55/// @brief Controls how grid data is read.
56///
57/// Passed via @c ReadOptions::readMode to influence which portions of a grid
58/// are deserialized and whether any on-the-fly conversion is performed during read.
59/// Codecs are expected to honour this setting in their @c readTopology() and
60/// @c readBuffers() implementations.
61enum class ReadMode {
62 /// Deserialize both topology and all value buffers. This is the default
63 /// behaviour and produces a fully populated grid identical to the one
64 /// that was written.
65 Original,
66 /// Deserialize topology and value buffers, converting each voxel value
67 /// to half-precision floating point on the fly. Codecs that do not
68 /// support native half conversion should fall back to @c Original and
69 /// record a @c ReadDiagnostic warning.
70 Half,
71 /// Deserialize topology and value buffers, converting each voxel value
72 /// to @c bool (non-zero to @c true, zero to @c false). Produces a
73 /// @c BoolGrid whose active set matches the source grid's active set.
74 Bool,
75 /// Deserialize topology only and promote the result to a mask grid,
76 /// discarding all value data. Equivalent to reading @c TopologyOnly and
77 /// then constructing a @c MaskGrid from the active-voxel set, but may
78 /// be performed more efficiently inside the codec.
79 Mask,
80 /// Deserialize topology only; reading of value buffers may be skipped.
81 /// Useful when only the active-voxel mask is needed and avoiding the cost
82 /// of reading large value buffers is desirable.
83 ///
84 /// @warning Only the topology is guaranteed. The resulting grid has a
85 /// valid tree structure (node hierarchy, tiles and active/inactive state),
86 /// but the voxel and tile values are unspecified. They are typically the
87 /// grid's background value, however a grid read from a file without grid
88 /// offsets is fully cached up-front, so the returned grid keeps the values
89 /// that were read. Deep-copying such a grid to reset its values would cost
90 /// more than the read this mode is meant to avoid. Callers must not rely
91 /// on the values being the background value.
92 TopologyOnly,
93 /// Deserialize grid metadata and transform only; no topology, no value
94 /// buffers. The codec is still used to construct the correct grid type,
95 /// but its @c readTopology()/@c readBuffers() are not called.
96 MetadataOnly
97};

99/// @brief Base class for per-grid-type, codec-specific read options.
100///
101/// Codecs that require type-specific read configuration should derive from
102/// this class and store an instance in @c ReadOptions::typeData, keyed by
103/// the grid type string (e.g. @c "Vec3SGrid").
104///
105/// @par Example
106/// @code
107/// struct MyCodecReadOptions : public ReadTypedOptions {
108/// bool someFlag = false;
109/// };
110///
111/// ReadOptions options;
112/// auto data = std::make_shared<MyCodecReadOptions>();
113/// data->someFlag = true;
114/// options.typeData["Vec3SGrid"] = data;
115/// @endcode
116struct OPENVDB_API ReadTypedOptions
117{
118 using Ptr = std::shared_ptr<ReadTypedOptions>;
119
120 virtual ~ReadTypedOptions() = default;
121
122 /// @brief Downcast a @c ReadTypedOptions::Ptr to a concrete derived type.
123 ///
124 /// Performs a @c static_cast to @c T after verifying via @c dynamic_cast
125 /// (when @c OPENVDB_ASSERT is enabled) that @p data actually holds an
126 /// instance of @c T. Calling this with a mismatched type is undefined
127 /// behaviour in release builds.
128 ///
129 /// @tparam T The concrete @c ReadTypedOptions subclass to cast to.
130 /// @param data A shared pointer to the base @c ReadTypedOptions object.
131 /// @return A reference to the underlying @c T instance.
132 template<typename T>
133 static T& cast(const Ptr& data) {
134 OPENVDB_ASSERT(dynamic_cast<T*>(data.get()) != nullptr);
135 return *static_cast<T*>(data.get());
136 }
137}; // struct ReadTypedOptions

139/// @brief Global read configuration passed to every codec during deserialization.
140///
141/// An instance of this struct is threaded through all codec @c readTopology() and
142/// @c readBuffers() calls so that site-wide policy (clipping, read mode) and any
143/// type-specific overrides are available in a single place.
144///
145/// @par Clipping
146/// When @c clipBBox is non-empty (i.e. @c !clipBBox.empty()), codecs should
147/// restrict the voxel data they load to the region that intersects that world-space
148/// bounding box. An empty bbox (the default) means no clipping is applied.
149///
150/// @par Read mode
151/// @c readMode controls the granularity and on-the-fly conversion applied
152/// during deserialization. The default, @c ReadMode::Original, loads both
153/// topology and value buffers without modification.
154/// @c ReadMode::Half, @c ReadMode::Bool, and @c ReadMode::Mask request
155/// in-place type conversion as data is read.
156/// @c ReadMode::TopologyOnly skips value buffers entirely, which can be
157/// significantly faster when only the active-voxel mask is needed.
158/// @c ReadMode::MetadataOnly skips both topology and value buffers,
159/// reading only grid metadata and transform.
160///
161/// @par Per-type options
162/// @c typeData allows callers to attach codec-specific configuration for
163/// individual grid types. Entries are keyed by the grid type string
164/// (e.g. @c "Vec3SGrid") and hold a @c ReadTypedOptions-derived object.
165/// Codecs retrieve their entry via @c ReadTypedOptions::cast<T>().
166struct OPENVDB_API ReadOptions
167{
168 /// World-space bounding box used to spatially clip the read.
169 /// An empty bbox (the default) disables clipping.
170 BBoxd clipBBox = BBoxd();
171
172 /// Controls which portions of each grid are deserialized.
173 /// Defaults to @c ReadMode::Original (full topology + value buffers).
174 ReadMode readMode = ReadMode::Original;
175
176 /// Optional per-grid-type codec configuration, keyed by grid type string.
177 /// Values are @c ReadTypedOptions subclass instances; use
178 /// @c ReadTypedOptions::cast<T>() to retrieve the concrete type.
179 std::unordered_map<std::string, ReadTypedOptions::Ptr> typeData;
180}; // struct ReadOptions

204/// @brief Thread-safe collection of read diagnostics accumulated during a codec read.
205///
206/// @details Diagnostics report situations where a requested read option could
207/// not be honoured by the codec — for example, clipping that was not applied
208/// natively and had to fall back to a post-process, or an option that the codec
209/// ignores entirely. The object must be explicitly enabled before any messages
210/// are recorded; it is disabled by default.
211struct OPENVDB_API ReadDiagnostics {
212 ReadDiagnostics() = default;
213 ReadDiagnostics(const ReadDiagnostics& other) {
214 std::lock_guard<std::mutex> lock(other.mMutex);
215 mEnabled = other.mEnabled;
216 mDiagnostics = other.mDiagnostics;
217 }
218 ReadDiagnostics& operator=(const ReadDiagnostics& other) {
219 if (this != &other) {
220 std::scoped_lock lock(mMutex, other.mMutex);
221 mEnabled = other.mEnabled;
222 mDiagnostics = other.mDiagnostics;
223 }
224 return *this;
225 }
226 /// @brief Append a warning diagnostic. No-op if the object is disabled.
227 /// @note This method is thread-safe.
228 void addWarning(const std::string& context, const std::string& message) {
229 if (!mEnabled) return;
230 std::lock_guard<std::mutex> lock(mMutex);
231 mDiagnostics.push_back({DiagnosticSeverity::Warning, context, message});
232 }
233 /// @brief Remove all recorded diagnostics.
234 /// @note This method is thread-safe.
235 void clear() {
236 std::lock_guard<std::mutex> lock(mMutex);
237 mDiagnostics.clear();
238 }
239 /// @brief Enable diagnostic recording.
240 void enable() { mEnabled = true; }
241 /// @brief Disable diagnostic recording.
242 void disable() { mEnabled = false; }
243 /// @brief Return @c true if diagnostic recording is enabled.
244 bool enabled() const { return mEnabled; }
245 /// @brief Return the list of recorded diagnostics.
246 const std::vector<ReadDiagnostic>& diagnostics() const { return mDiagnostics; }
247 /// @brief Return @c true if no diagnostics have been recorded.
248 bool empty() const { return mDiagnostics.empty(); }
249private:
250 bool mEnabled = false;
251 std::vector<ReadDiagnostic> mDiagnostics;
252 mutable std::mutex mMutex;
253}; // struct ReadDiagnostics

255/// @brief Mutable per-operation state created by @c Codec::createData() and
256/// passed into every read and write call on a given codec.
257///
258/// @details The main purpose of this class is to provide a mechanism for passing
259/// additional state between the different methods in the codec. @c ReadOptions is
260/// immutable and cannot carry per-operation state; the codec itself is stateless
261/// by design (to allow sharing across threads) and therefore also cannot be used
262/// for this purpose.
263///
264/// The base class provides the single field that every codec requires:
265/// a shared pointer to the @c GridBase being populated (on read) or inspected
266/// (on write). Derived codecs that need additional transient state should
267/// subclass @c CodecData and add those fields.
268///
269/// A @c CodecData instance is created once per grid per read or write operation
270/// by @c Codec::createData(), which also allocates the concrete @c GridBase
271/// subclass and stores it in @c grid.
272///
273/// @par Deriving from CodecData
274/// @code
275/// struct MyCodecData : public io::CodecData {
276/// // Additional per-operation state goes here
277/// SomeCache intermediateBuffer;
278/// };
279/// @endcode
280///
281/// Inside @c Codec::createData(), the derived type is allocated and returned:
282/// @code
283/// io::CodecData::Ptr createData() override {
284/// auto data = std::make_unique<MyCodecData>();
285/// data->grid = MyGridType::create();
286/// return data;
287/// }
288/// @endcode
289///
290/// In the read and write methods, downcast the reference back to the concrete
291/// type to access the extra fields:
292/// @code
293/// void readBuffers(std::istream& is, int64_t size, io::CodecData& data, ...) const override {
294/// auto& myData = static_cast<MyCodecData&>(data);
295/// MyGridType& grid = static_cast<MyGridType&>(*myData.grid);
296/// // Use myData.intermediateBuffer, grid, etc.
297/// }
298/// @endcode
299struct OPENVDB_API CodecData
300{
301 using Ptr = std::unique_ptr<CodecData>;
302
303 virtual ~CodecData() = default;
304
305 /// The grid being populated on read, or being serialized on write.
306 SharedPtr<GridBase> grid;
307}; // struct CodecData

309/// @brief Abstract base class for grid I/O codecs.
310///
311/// @details A codec encapsulates the serialization and deserialization logic
312/// for a specific grid type (or family of grid types). The I/O subsystem
313/// selects a codec by name at runtime, calls @c createData() to allocate
314/// per-operation state, and then dispatches to the appropriate read or write
315/// methods.
316///
317/// Codecs must be stateless: all mutable per-operation state is stored in a
318/// @c CodecData object (see above) that is created fresh for each grid read
319/// or write. This ensures that a single registered codec instance can safely
320/// handle concurrent I/O operations.
321///
322/// @par Implementing a new codec
323///
324/// 1. **Declare the codec struct**, deriving publicly from @c Codec (or from a
325/// convenience intermediate that already handles topology, if one exists for
326/// the grid family in question):
327/// @code
328/// struct MyCodec : public io::Codec { ... };
329/// @endcode
330///
331/// 2. **Provide a static @c name() method** that returns the unique string
332/// identifier under which the codec will be registered. Any globally
333/// unique ASCII string is acceptable:
334/// @code
335/// static std::string name() { return "mycodec"; }
336/// @endcode
337///
338/// 3. **Override @c createData()** (the only pure-virtual method). Allocate a
339/// @c CodecData (or a derived subclass if extra per-operation state is
340/// needed), create the appropriate @c GridBase subclass, assign it to
341/// @c CodecData::grid, and return the object:
342/// @code
343/// io::CodecData::Ptr createData() override {
344/// auto data = std::make_unique<io::CodecData>();
345/// data->grid = MyGridType::create();
346/// return data;
347/// }
348/// @endcode
349///
350/// 4. **Override the read and/or write methods** as required. All four
351/// methods have default no-op implementations, so override only those that
352/// the codec actually uses. The I/O layer separates topology (the tree
353/// structure and active/inactive voxel mask) from value buffers (the actual
354/// voxel data), allowing callers to request topology-only reads via
355/// @c ReadOptions::readMode. Implementations should honour that setting:
356/// @code
357/// void readTopology(std::istream& is, io::CodecData& data,
358/// const io::ReadOptions& options,
359/// io::ReadDiagnostics& diagnostics) const override
360/// {
361/// MyGridType& grid = static_cast<MyGridType&>(*data.grid);
362/// // Deserialize the tree structure into grid...
363/// }
364///
365/// void readBuffers(std::istream& is, int64_t size, CodecData& data,
366/// const io::ReadOptions& options,
367/// io::ReadDiagnostics& diagnostics) const override
368/// {
369/// if (options.readMode == io::ReadMode::TopologyOnly) return;
370/// MyGridType& grid = static_cast<MyGridType&>(*data.grid);
371/// // Deserialize voxel values into grid...
372/// }
373/// @endcode
374/// If a requested option (e.g. spatial clipping) cannot be honoured natively,
375/// record a warning via @c diagnostics.addWarning() rather than silently ignoring it.
376///
377/// A grid is read in four phases:
378/// @c beginRead() => @c readTopology() => @c readBuffers() => @c finishRead().
379///
380/// For non-instance grid reading, only @c beginRead and @c finishRead are guaranteed to run.
381/// A topology-only read will skip the @c readBuffers() phase for example. Compute tasks may
382/// continue asynchronously across phases, the codec owns that work and is responsible for
383/// starting and ending it.
384///
385/// 5. **Register the codec** once at start-up, typically from the library's
386/// @c initialize() function, using @c CodecRegistry::registerCodec<MyCodec>():
387/// @code
388/// io::CodecRegistry::registerCodec<MyCodec>();
389/// @endcode
390///
391/// @note The codec struct itself is never copied; the registry takes ownership
392/// of a single heap-allocated instance via @c Codec::Ptr
393/// (@c std::unique_ptr<Codec>).
394struct OPENVDB_API Codec
395{
396 using Ptr = std::unique_ptr<Codec>;
397
398 virtual ~Codec() = default;
399
400 /// @brief Allocate per-operation codec state, including the target grid.
401 ///
402 /// This is the only pure-virtual method. Implementations must create a
403 /// concrete @c GridBase subclass appropriate for the codec, store it in
404 /// @c CodecData::grid, and return the @c CodecData (or a derived subclass
405 /// carrying additional state). The returned object is passed by reference
406 /// to every subsequent read or write call for the same grid.
407 ///
408 /// @return A fully initialized @c CodecData whose @c grid field is non-null.
409 virtual CodecData::Ptr createData() = 0;
410
411 /// @brief Begin a read. Called by the I/O layer once before the first phase of every read
412 /// of a grid, for every read mode, after the metadata and the transform are read.
413 /// It is not called for instances, which have no phases.
414 ///
415 /// @details This is the place to start work that later phases use, for example a thread
416 /// pool, or to choose a strategy from the read options, the metadata and the transform.
417 /// Until @c finishRead() returns the caller does not touch the tree of the grid.
418 /// If this throws, no other phase is called, including @c finishRead(), so the
419 /// destructor of the @c CodecData must release anything started here. The default
420 /// implementation does nothing.
421 virtual void beginRead(CodecData& /*data*/, const ReadOptions& /*options*/,
422 ReadDiagnostics& /*diagnostics*/) const { }
423
424 /// @brief Deserialize the grid topology (tree structure and active-voxel
425 /// mask) from @a is into the grid held by @a data.
426 virtual void readTopology(std::istream& /*is*/, CodecData& /*data*/,
427 const ReadOptions& /*options*/, ReadDiagnostics& /*diagnostics*/) const { }
428
429 /// @brief Deserialize all voxel-value buffers from @a is into the grid
430 /// held by @a data.
431 ///
432 /// The default implementation is a no-op. Override to populate leaf-node
433 /// value buffers after topology has been established. When
434 /// @c options.readMode is @c ReadMode::TopologyOnly this method will not
435 /// be called by the I/O layer, so implementations may also guard against
436 /// that mode internally for safety. If @c options.clipBBox is non-empty,
437 /// restrict the loaded data to the region that intersects it; if the codec
438 /// cannot honour clipping natively, fall back to a post-process and record
439 /// a warning via @a diagnostics. The @c size argument is the number of bytes
440 /// occupied by the readBuffers data section, measured from the stream
441 /// position at which this method is called, or -1 when the size is not known
442 /// (a non-seekable stream, or a stream without grid offsets).
443 virtual void readBuffers(std::istream& /*is*/, int64_t /*size*/, CodecData& /*data*/,
444 const ReadOptions& /*options*/, ReadDiagnostics& /*diagnostics*/) const { }
445
446 /// @brief Finish a read. Called by the I/O layer after the last phase of every read of a
447 /// grid, for every read mode and whether or not @c readTopology() and @c readBuffers()
448 /// were called, so work that a codec started in an earlier phase has one place to be
449 /// joined. It is not called for instances, which have no phases.
450 ///
451 /// @details Until this returns the caller does not touch the tree of the grid. It may read
452 /// and change the metadata and the transform. An error from work started in an earlier
453 /// phase is thrown from here, on the calling thread. The default implementation does
454 /// nothing.
455 virtual void finishRead(CodecData& /*data*/, const ReadOptions& /*options*/,
456 ReadDiagnostics& /*diagnostics*/) const { }
457
458 /// @brief Serialize the grid topology (tree structure and active-voxel
459 /// mask) from @a grid to @a os.
460 ///
461 /// The default implementation is a no-op. Override when the codec stores
462 /// topology as a distinct section that precedes the value buffers in the
463 /// stream.
464 virtual void writeTopology(std::ostream& /*os*/, const GridBase& /*grid*/,
465 const WriteOptions& /*options*/) const { }
466
467 /// @brief Serialize all voxel-value buffers from @a grid to @a os.
468 ///
469 /// The default implementation is a no-op. Override to write the leaf-node
470 /// value buffers that follow the topology section.
471 virtual void writeBuffers(std::ostream& /*os*/, const GridBase& /*grid*/,
472 const WriteOptions& /*options*/) const { }
473}; // struct Codec

475/// @brief A thread-safe, process-global registry that maps codec names to
476/// @c Codec instances.
477///
478/// Codecs are identified by a unique string name (typically provided by @c Codec::name())
479/// and stored as @c Codec::Ptr (i.e. @c std::unique_ptr<Codec>).
480///
481/// @par Typical usage
482/// @code
483/// // Register a custom codec once at start-up (e.g. from initialize()):
484/// CodecRegistry::registerCodec<MyCodec>();
485/// @endcode
486///
487/// @note Attempting to register a codec whose name is already present throws
488/// @c openvdb::KeyError. Call @c isRegistered() first when the caller
489/// cannot guarantee uniqueness.
490struct OPENVDB_API CodecRegistry
491{
492 /// Return @c true if a codec with the given @a name has been registered.
493 static bool isRegistered(const std::string& name);
494
495 /// Register a codec under the explicit string @a name, transferring
496 /// ownership of @a codec to the registry.
497 /// @throw KeyError if @a name is already registered.
498 static void registerCodecByName(const std::string& name, Codec::Ptr&& codec);
499
500 /// Convenience wrapper that registers @c CodecT using the name returned
501 /// by @c CodecT::name(), constructing the instance internally.
502 /// @throw KeyError if the codec is already registered.
503 template <typename CodecT>
504 static void registerCodec()
505 {
506 registerCodecByName(CodecT::name(), std::make_unique<CodecT>());
507 }
508
509 /// Return a raw (non-owning) pointer to the codec registered under
510 /// @a name, or @c nullptr if no such codec exists.
511 /// The returned pointer remains valid for the lifetime of the registry.
512 static Codec* get(const std::string& name);
513
514 /// Deregister all codecs and reset the registry to an empty state.
515 static void clear();
516}; // struct CodecRegistry
