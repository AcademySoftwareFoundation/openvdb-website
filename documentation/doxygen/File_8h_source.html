29/// Grid archive associated with a file on disk
30class OPENVDB_API File: public Archive
31{
32public:
33 using NameMap = std::multimap<Name, GridDescriptor>;
34 using NameMapCIter = NameMap::const_iterator;
35
36 explicit File(const std::string& filename);
37 ~File() override { }
38
39 /// @brief Copy constructor
40 /// @details The copy will be closed and will not reference the same
41 /// file descriptor as the original.
42 File(const File& other);
43 /// @brief Assignment
44 /// @details After assignment, this File will be closed and will not
45 /// reference the same file descriptor as the source File.
46 File& operator=(const File& other);

48 /// @brief Return a copy of this archive.
49 /// @details The copy will be closed and will not reference the same
50 /// file descriptor as the original.
51 SharedPtr<Archive> copy() const override;

53 /// @brief Return the name of the file with which this archive is associated.
54 /// @details The file does not necessarily exist on disk yet.
55 const std::string& filename() const;
56
57 /// @brief Open the file, read the file header and the file-level metadata,
58 /// and populate the grid descriptors, but do not load any grids into memory.
59 /// @throw IoError if the file is not a valid VDB file.
60 /// @return @c true if the file's UUID has changed since it was last read.
61 bool open();

68 /// Return @c true if the file has been opened for reading.
69 bool isOpen() const;
70
71 /// Close the file once we are done reading from it.
72 void close();
73
74 /// @brief Return this file's current size on disk in bytes.
75 /// @throw IoError if the file size cannot be determined.
76 Index64 getSize() const;
77
78 OPENVDB_DEPRECATED_MESSAGE("Always returns 0. This method is deprecated and will be removed. Delayed loading is no longer supported.")
79 Index64 copyMaxBytes() const { return 0; }
80 OPENVDB_DEPRECATED_MESSAGE("This method is deprecated and will be removed. Delayed loading is no longer supported.")
81 void setCopyMaxBytes(Index64 /*bytes*/) { }

83 /// Return @c true if a grid of the given name exists in this file.
84 bool hasGrid(const Name&) const;
85
86 /// Return (in a newly created MetaMap) the file-level metadata.
87 MetaMap::Ptr getMetadata() const;
88
89 /// @brief Read the entire contents of the file and return a list of grid pointers.
90 /// @details Instances share trees, except when clipping a file that has grid offsets,
91 /// where each grid is clipped independently.
92 GridPtrVecPtr getGrids(const io::ReadOptions& readOptions = io::ReadOptions{}) const;
93
94 /// @brief Read just the grid metadata and transforms from the file and return a list
95 /// of pointers to grids that are empty except for their metadata and transforms.
96 /// @throw IoError if this file is not open for reading.
97 GridPtrVecPtr readAllGridMetadata();
98
99 /// @brief Read a grid's metadata and transform only.
100 /// @return A pointer to a grid that is empty except for its metadata and transform.
101 /// @throw IoError if this file is not open for reading.
102 /// @throw KeyError if no grid with the given name exists in this file.
103 GridBase::Ptr readGridMetadata(const Name&);
104
105 /// Read an entire grid, including all of its data blocks, using the provided options if given.
106 GridBase::Ptr readGrid(const Name&, const io::ReadOptions& readOptions = io::ReadOptions{});
107 /// @brief Read a grid, including its data blocks, but only where it
108 /// intersects the given world-space bounding box.
109 GridBase::Ptr readGrid(const Name&, const BBoxd&);
110
111 /// @brief Read all grids whose grid name is exactly @a name, in file order.
112 /// @details Grids have a name, such as "density", which may not be unique in a file.
113 /// When more than one grid share the same name, these names are made unique by adding
114 /// a [N] suffix. For example, the first "density" grid becomes "density[0]", the second
115 /// becomes "density[1]", etc. This method ignores the unique name and returns all grids
116 /// whose grid name exactly matches the given name.
117 /// Instances share trees as they do with getGrids(), except that a clip on a file
118 /// with grid offsets does a clip on each grid independently.
119 /// @return An empty list if no grid has this name.
120 /// @throw IoError if this file is not open for reading.
121 GridPtrVecPtr readGrids(const Name& name, const io::ReadOptions& readOptions = io::ReadOptions{});
122
123 /// @brief Write the grids in the given container to the file whose name
124 /// was given in the constructor.
125 void write(const GridCPtrVec&, const MetaMap& = MetaMap(),
126 const io::WriteOptions& = io::WriteOptions{}) const override;
127
128 /// @brief Write the grids in the given container to the file whose name
129 /// was given in the constructor.
130 template<typename GridPtrContainerT>
131 void write(const GridPtrContainerT&, const MetaMap& = MetaMap(),
132 const io::WriteOptions& = io::WriteOptions{}) const;

134 /// A const iterator that iterates over all names in the file. This is only
135 /// valid once the file has been opened.
136 class OPENVDB_API NameIterator
137 {
138 public:
139 NameIterator(const NameMapCIter& iter): mIter(iter) {}
140 NameIterator(const NameIterator&) = default;
141 ~NameIterator() {}
142
143 NameIterator& operator++() { mIter++; return *this; }
144
145 bool operator==(const NameIterator& iter) const { return mIter == iter.mIter; }
146 bool operator!=(const NameIterator& iter) const { return mIter != iter.mIter; }
147
148 Name operator*() const { return this->gridName(); }
149
150 Name gridName() const { return GridDescriptor::nameAsString(mIter->second.uniqueName()); }
151
152 private:
153 NameMapCIter mIter;
154 };
155
156 /// @return a NameIterator to iterate over all grid names in the file.
157 NameIterator beginName() const;
158
159 /// @return the ending iterator for all grid names in the file.
160 NameIterator endName() const;
161
162private:
163 /// @brief Return an iterator to the descriptor for the grid with the given name.
164 /// @details If "density" is searched and there is more than one "density" grid,
165 /// returns an iterator to the first matching grid. If "density[1]" is searched,
166 /// it is tested against the unique name, so it matches the second "density" grid.
167 NameMapCIter findDescriptor(const Name&) const;
168
169 /// @brief Return an iterator to the descriptor whose unique name is exactly the given name,
170 /// or the end iterator if there is none.
171 /// @details Only the unique name is searched. "density" does not match a grid that was
172 /// made unique as "density[1]", "density[1]" must be used instead.
173 NameMapCIter findDescriptorByUniqueName(const Name&) const;
174
175 /// @brief Read the grid for the given descriptor, including its instance parent's tree.
176 /// @note This method should not be called for files that don't contain grid offsets.
177 /// @throw KeyError if the grid is an instance and its parent is missing.
178 GridBase::Ptr readGridFromDescriptor(const GridDescriptor&, const io::ReadOptions&) const;
179
180 /// @brief Retrieve a grid from @c mNamedGrids. Return a null pointer
181 /// if @c mNamedGrids was not populated (because this file is random-access).
182 /// @throw KeyError if no grid with the given name exists in this file.
183 GridBase::Ptr retrieveCachedGrid(const Name&) const;
184
185 /// @brief Return the grid that @a readOptions asks for, given the already
186 /// cached @a cachedGrid. Doesn't mutate @a cachedGrid, and returns it
187 /// unchanged when no option applies.
188 GridBase::Ptr resolveCachedGrid(const GridBase::Ptr& cachedGrid,
189 const io::ReadOptions& readOptions, ReadDiagnostics& diagnostics) const;
190
191 /// @brief Resolve each of @a cachedGrids as resolveCachedGrid() does. Grids that
192 /// share a source tree share the resolved tree when instancing is enabled.
193 GridPtrVecPtr resolveCachedGrids(const GridPtrVec& cachedGrids,
194 const io::ReadOptions& readOptions) const;
195
196 void writeGrids(const GridCPtrVec&, const MetaMap&, const io::WriteOptions&) const;
197
198 MetaMap::Ptr fileMetadata();
199 MetaMap::ConstPtr fileMetadata() const;
200
201 const NameMap& gridDescriptors() const;
202 NameMap& gridDescriptors();
203
204 std::istream& inputStream() const;
205
206 friend class ::TestFile;
207 friend class ::TestStream;
208
209 std::string mFilename;
210 // The file-level metadata
211 MetaMap::Ptr mMeta;
212 // The file stream that is open for reading
213 std::unique_ptr<std::istream> mInStream;
214 // File-level stream metadata (file format, compression, etc.)
215 StreamMetadata::Ptr mStreamMetadata;
216 // Flag indicating if we have read in the global information (header,
217 // metadata, and grid descriptors) for this VDB file
218 bool mIsOpen = false;
219 // Grid descriptors for all grids stored in the file, indexed by grid name
220 NameMap mGridDescriptors;
221 // All grids, indexed by unique name (used only when mHasGridOffsets is false)
222 Archive::NamedGridMap mNamedGrids;
223 // All grids stored in the file (used only when mHasGridOffsets is false)
224 GridPtrVecPtr mGrids;
225};

231inline void
232File::write(const GridCPtrVec& grids, const MetaMap& meta,
233 const io::WriteOptions& writeOptions) const
234{
235 this->writeGrids(grids, meta, writeOptions);
236}

239template<typename GridPtrContainerT>
240inline void
241File::write(const GridPtrContainerT& container, const MetaMap& meta,
242 const io::WriteOptions& writeOptions) const
243{
244 GridCPtrVec grids;
245 std::copy(container.begin(), container.end(), std::back_inserter(grids));
246 this->writeGrids(grids, meta, writeOptions);
247}
