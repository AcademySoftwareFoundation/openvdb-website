20/// Grid archive associated with arbitrary input and output streams (not necessarily files)
21class OPENVDB_API Stream: public Archive
22{
23public:
24 /// @brief Read grids from an input stream.
25 /// @param is The input stream to read from
26 explicit Stream(std::istream& is);
27
28 /// @brief Read grids from an input stream using the given read options.
29 /// @param is The input stream to read from
30 /// @param readOptions Options controlling how grids are read (e.g. attribute
31 /// skipping for point data grids)
32 /// @throw ValueError if @a readOptions requests @c io::ReadMode::MetadataOnly
33 /// or @c io::ReadMode::TopologyOnly. These modes leave a grid's data
34 /// partially unread, which a sequentially read stream cannot skip over.
35 Stream(std::istream& is, const io::ReadOptions& readOptions);
36
37 OPENVDB_DEPRECATED_MESSAGE("Use Stream(std::istream&) instead. This method is deprecated and will be removed. Delayed loading is no longer supported.")
38 Stream(std::istream& is, bool /*delayLoad*/) : Stream(is) { }
39
40 /// Construct an archive for stream output.
41 Stream() = default;
42 /// Construct an archive for output to the given stream.
43 explicit Stream(std::ostream&);
44
45 Stream(const Stream&);
46 Stream& operator=(const Stream&);
47
48 ~Stream() override { }
49
50 /// @brief Return a copy of this archive.
51 Archive::Ptr copy() const override;
52
53 /// Return the file-level metadata in a newly created MetaMap.
54 MetaMap::Ptr getMetadata() const;
55
56 /// Return pointers to the grids that were read from the input stream.
57 GridPtrVecPtr getGrids();
58
59 /// @brief Write the grids in the given container to this archive's output stream.
60 /// @throw ValueError if this archive was constructed without specifying an output stream.
61 void write(const GridCPtrVec&, const MetaMap& = MetaMap(),
62 const io::WriteOptions& = io::WriteOptions{}) const override;
63
64 /// @brief Write the grids in the given container to this archive's output stream.
65 /// @throw ValueError if this archive was constructed without specifying an output stream.
66 template<typename GridPtrContainerT>
67 void write(const GridPtrContainerT&, const MetaMap& = MetaMap(),
68 const io::WriteOptions& = io::WriteOptions{}) const;
69
70private:
71 void writeGrids(std::ostream&, const GridCPtrVec&, const MetaMap&,
72 const io::WriteOptions&) const;
73
74 MetaMap::Ptr mMeta;
75 GridPtrVecPtr mGrids;
76 std::ostream* mOutputStream = nullptr;
77};

83template<typename GridPtrContainerT>
84inline void
85Stream::write(const GridPtrContainerT& container, const MetaMap& metadata,
86 const io::WriteOptions& writeOptions) const
87{
88 GridCPtrVec grids;
89 std::copy(container.begin(), container.end(), std::back_inserter(grids));
90 this->write(grids, metadata, writeOptions);
91}
