23/// @brief openvdb::io::readCompressedValues specialized on PointDataIndex32 arrays to
24/// ignore the value mask, use a larger block size and use 16-bit size instead of 64-bit
25template<>
26inline void
27readCompressedValues( std::istream& is, PointDataIndex32* destBuf, Index destCount,
28 const util::NodeMask<3>& /*valueMask*/, bool /*fromHalf*/,
29 const PointDataIndex32* /*background*/)
30{
31 using compression::bloscDecompress;
32
33 const bool seek = destBuf == nullptr;
34
35 const size_t destBytes = destCount*sizeof(PointDataIndex32);
36 const size_t maximumBytes = std::numeric_limits<uint16_t>::max();
37 if (destBytes >= maximumBytes) {
38 OPENVDB_THROW(openvdb::IoError, "Cannot read more than " <<
39 maximumBytes << " bytes in voxel values.")
40 }
41
42 uint16_t bytes16;
43
44 const io::StreamMetadata::Ptr meta = io::getStreamMetadataPtr(is);
45
46 if (seek && meta) {
47 // buffer size temporarily stored in the StreamMetadata pass
48 // to avoid having to perform an expensive disk read for 2-bytes
49 bytes16 = static_cast<uint16_t>(meta->pass());
50 // seek over size of the compressed buffer
51 is.seekg(sizeof(uint16_t), std::ios_base::cur);
52 }
53 else {
54 // otherwise read from disk
55 is.read(reinterpret_cast<char*>(&bytes16), sizeof(uint16_t));
56 }
57
58 if (bytes16 == std::numeric_limits<uint16_t>::max()) {
59 // read or seek uncompressed data
60 if (seek) {
61 is.seekg(destBytes, std::ios_base::cur);
62 }
63 else {
64 is.read(reinterpret_cast<char*>(destBuf), destBytes);
65 }
66 }
67 else {
68 // read or seek uncompressed data
69 if (seek) {
70 is.seekg(int(bytes16), std::ios_base::cur);
71 }
72 else {
73 // decompress into the destination buffer
74 std::unique_ptr<char[]> bloscBuffer(new char[int(bytes16)]);
75 is.read(bloscBuffer.get(), bytes16);
76 compression::bloscValidateCompressedBytes(bloscBuffer.get(), size_t(bytes16));
77 std::unique_ptr<char[]> buffer = bloscDecompress( bloscBuffer.get(),
78 destBytes,
79 /*resize=*/false);
80 std::memcpy(destBuf, buffer.get(), destBytes);
81 }
82 }
83}

85/// @brief openvdb::io::writeCompressedValues specialized on PointDataIndex32 arrays to
86/// ignore the value mask, use a larger block size and use 16-bit size instead of 64-bit
87template<>
88inline void
89writeCompressedValues( std::ostream& os, const PointDataIndex32* srcBuf, Index srcCount,
90 const util::NodeMask<3>& /*valueMask*/,
91 const util::NodeMask<3>& /*childMask*/, bool /*toHalf*/,
92 const PointDataIndex32* /*background*/)
93{
94 using compression::bloscCompress;
95
96 const size_t srcBytes = srcCount*sizeof(PointDataIndex32);
97 const size_t maximumBytes = std::numeric_limits<uint16_t>::max();
98 if (srcBytes >= maximumBytes) {
99 OPENVDB_THROW(openvdb::IoError, "Cannot write more than " <<
100 maximumBytes << " bytes in voxel values.")
101 }
102
103 const char* charBuffer = reinterpret_cast<const char*>(srcBuf);
104
105 size_t compressedBytes;
106 std::unique_ptr<char[]> buffer = bloscCompress( charBuffer, srcBytes,
107 compressedBytes, /*resize=*/false);
108
109 if (compressedBytes > 0) {
110 auto bytes16 = static_cast<uint16_t>(compressedBytes); // clamp to 16-bit unsigned integer
111 os.write(reinterpret_cast<const char*>(&bytes16), sizeof(uint16_t));
112 os.write(reinterpret_cast<const char*>(buffer.get()), compressedBytes);
113 }
114 else {
115 auto bytes16 = static_cast<uint16_t>(maximumBytes); // max value indicates uncompressed
116 os.write(reinterpret_cast<const char*>(&bytes16), sizeof(uint16_t));
117 os.write(reinterpret_cast<const char*>(srcBuf), srcBytes);
118 }
119}

121template <typename T>
122inline void
123writeCompressedValuesSize(std::ostream& os, const T* srcBuf, Index srcCount)
124{
125 using compression::bloscCompressedSize;
126
127 const size_t srcBytes = srcCount*sizeof(T);
128 const size_t maximumBytes = std::numeric_limits<uint16_t>::max();
129 if (srcBytes >= maximumBytes) {
130 OPENVDB_THROW(openvdb::IoError, "Cannot write more than " <<
131 maximumBytes << " bytes in voxel values.")
132 }
133
134 const char* charBuffer = reinterpret_cast<const char*>(srcBuf);
135
136 // calculate voxel buffer size after compression
137 size_t compressedBytes = bloscCompressedSize(charBuffer, srcBytes);
138
139 if (compressedBytes > 0) {
140 auto bytes16 = static_cast<uint16_t>(compressedBytes); // clamp to 16-bit unsigned integer
141 os.write(reinterpret_cast<const char*>(&bytes16), sizeof(uint16_t));
142 }
143 else {
144 auto bytes16 = static_cast<uint16_t>(maximumBytes); // max value indicates uncompressed
145 os.write(reinterpret_cast<const char*>(&bytes16), sizeof(uint16_t));
146 }
147}
