30PNANOVDB_IO_INLINE void* pnanovdb_aligned_malloc(size_t alignment, size_t size)
31{
32 size_t aligned_size = size + alignment + sizeof(void*);
33 void* alloc_ptr = malloc(aligned_size);
34 void* ptr = 0;
35 if (alloc_ptr)
36 {
37 ptr = (void*)(alignment * (((size_t)alloc_ptr + sizeof(void*) + alignment - 1u) / alignment));
38 void** header_ptr = (void**)((size_t)ptr - sizeof(void*));
39 *header_ptr = alloc_ptr;
40 }
41 return ptr;
42}
43PNANOVDB_IO_INLINE void pnanovdb_aligned_free(void* ptr)
44{
45 if (ptr)
46 {
47 void** header_ptr = (void**)((size_t)ptr - sizeof(void*));
48 free(*header_ptr);
49 }

72// implementation for dynamic buffer
73typedef struct pnanovdb_io_dynamic_buf_t
74{
75 void* data;
76 uint64_t size;
77 uint64_t capacity;
78}pnanovdb_io_dynamic_buf_t;
79PNANOVDB_IO_INLINE void pnanovdb_io_dynamic_buf_read(void* file, uint64_t offset, void* dst_data, uint64_t size)
80{
81 pnanovdb_io_dynamic_buf_t* ptr = (pnanovdb_io_dynamic_buf_t*)file;
82 uint8_t* dst8 = (uint8_t*)dst_data;
83 uint8_t* src8 = (uint8_t*)ptr->data;
84 uint64_t size_in_bounds = size;
85 if (offset >= ptr->size)
86 {
87 size_in_bounds = 0;
88 }
89 else if (offset + size > ptr->size)
90 {
91 size_in_bounds = ptr->size - offset;
92 }
93 memcpy(dst8, src8 + offset, size_in_bounds);
94 memset(dst8 + size_in_bounds, 0, size - size_in_bounds);
95}
96PNANOVDB_IO_INLINE void pnanovdb_io_dynamic_buf_write(void* file, uint64_t offset, void* src_data, uint64_t size)
97{
98 pnanovdb_io_dynamic_buf_t* ptr = (pnanovdb_io_dynamic_buf_t*)file;
99 // grow capacity as needed
100 if (offset + size > ptr->capacity)
101 {
102 uint64_t new_capacity = 1024u;
103 while (new_capacity < offset + size)
104 {
105 new_capacity = (new_capacity << 1u) - (new_capacity >> 1u);
106 }
107 void* new_data = pnanovdb_aligned_malloc(PNANOVDB_DATA_ALIGNMENT, new_capacity);
108 memcpy(new_data, ptr->data, ptr->size);
109 if (ptr->capacity != 0u)
110 {
111 pnanovdb_aligned_free(ptr->data);
112 }
113 ptr->data = new_data;
114 ptr->capacity = new_capacity;
115 }
116 // expand size and zero initialize as needed
117 if (offset + size > ptr->size)
118 {
119 uint8_t* data8 = (uint8_t*)ptr->data;
120 memset(data8 + ptr->size, 0, offset + size - ptr->size);
121 ptr->size = offset + size;
122 }
123 // write data
124 uint8_t* src8 = (uint8_t*)src_data;
125 uint8_t* dst8 = (uint8_t*)ptr->data;
126 memcpy(dst8 + offset, src8, size);
127}

200PNANOVDB_IO_INLINE pnanovdb_grid_handle_t pnanovdb_io_fread_fileheader(
201 pnanovdb_buf_t file,
202 pnanovdb_io_fileheader_t* dst_header,
203 pnanovdb_io_filemetadata_t* dst_metadatas,
204 pnanovdb_uint64_t dst_metadata_count)
205{
206 pnanovdb_grid_t local_grid = {};
207 file->read(file->file, 0u, &local_grid, PNANOVDB_GRID_SIZE);
208 pnanovdb_bool_t is_raw_grid = PNANOVDB_FALSE;
209 if (local_grid.magic == PNANOVDB_MAGIC_GRID ||
210 local_grid.data2 == PNANOVDB_MAGIC_GRID)
211 {
212 is_raw_grid = PNANOVDB_TRUE;
213 }
214 if (is_raw_grid)
215 {
216 pnanovdb_grid_handle_t raw_grid = {pnanovdb_address_null()};
217 return raw_grid;
218 }
219
220 pnanovdb_uint64_t offset = 0u;
221 pnanovdb_io_fileheader_t local_header = {};
222 file->read(file->file, offset, &local_header, PNANOVDB_IO_FILEHEADER_SIZE);
223 if (dst_header)
224 {
225 *dst_header = local_header;
226 }
227 offset += PNANOVDB_IO_FILEHEADER_SIZE;
228 pnanovdb_uint32_t grid_count = local_header.grid_count_and_codec & 0xFFFF;
229 for (uint32_t idx = 0u; idx < grid_count; idx++)
230 {
231 pnanovdb_io_filemetadata_t local_metadata = {};
232 file->read(file->file, offset, &local_metadata, PNANOVDB_IO_FILEMETADATA_SIZE);
233 if (idx < dst_metadata_count)
234 {
235 dst_metadatas[idx] = local_metadata;
236 }
237 offset += PNANOVDB_IO_FILEMETADATA_SIZE;
238 // skip past name
239 offset += local_metadata.name_size;
240 }
241 pnanovdb_grid_handle_t first_grid = {pnanovdb_address_offset64(pnanovdb_address_null(), offset)};
242 return first_grid;
243}

245PNANOVDB_IO_INLINE void pnanovdb_io_memcpy(
246 pnanovdb_buf_t dst_buf, pnanovdb_address_t dst_addr,
247 pnanovdb_buf_t src_buf, pnanovdb_address_t src_addr,

254 pnanovdb_write_uint32(dst_buf, pnanovdb_address_offset_product(dst_addr, word_idx, 4u), val);
