30/// @brief Array of fixed size 2<SUP>3<I>Log2Dim</I></SUP> that stores
31/// the voxel values of a LeafNode
32template<typename T, Index Log2Dim>
33class LeafBuffer
34{
35public:
36 using ValueType = T;
37 using StorageType = ValueType;
38 using NodeMaskType = util::NodeMask<Log2Dim>;
39 static const Index SIZE = 1 << 3 * Log2Dim;
40
41 /// Default constructor
42 inline LeafBuffer(): mData(new ValueType[SIZE]) {}
43 /// Construct a buffer populated with the specified value.
44 explicit inline LeafBuffer(const ValueType&);
45 /// Copy constructor
46 inline LeafBuffer(const LeafBuffer&);
47 /// @brief Construct a buffer without allocating the value array.
48 /// The buffer is left unallocated; call @c allocate() before use.
49 /// Populating the buffer (via @c allocate() or @c data()) must be done
50 /// single-threaded per leaf, or externally synchronized. The valid
51 /// advanced pattern is: create all nodes single-threaded, then call
52 /// @c allocate() in parallel across distinct leaves.
53 LeafBuffer(PartialCreate, const ValueType&): mData(nullptr) {}
54 /// Destructor
55 inline ~LeafBuffer();
56
57 /// Return @c true if memory for this buffer has not yet been allocated.
58 bool empty() const { return !mData; }
59 /// Allocate memory for this buffer if it has not already been allocated.
60 bool allocate() { if (mData == nullptr) mData = new ValueType[SIZE]; return true; }
61
62 /// Populate this buffer with a constant value.
63 inline void fill(const ValueType&);
64
65 /// Return a const reference to the i'th element of this buffer.
66 const ValueType& getValue(Index i) const { return this->at(i); }
67 /// Return a const reference to the i'th element of this buffer.
68 const ValueType& operator[](Index i) const { return this->at(i); }
69 /// Set the i'th value of this buffer to the specified value.
70 inline void setValue(Index i, const ValueType&);
71
72 /// Copy the other buffer's values into this buffer.
73 inline LeafBuffer& operator=(const LeafBuffer&);
74
75 /// @brief Return @c true if the contents of the other buffer
76 /// exactly equal the contents of this buffer.
77 inline bool operator==(const LeafBuffer&) const;
78 /// @brief Return @c true if the contents of the other buffer
79 /// are not exactly equal to the contents of this buffer.
80 inline bool operator!=(const LeafBuffer& other) const { return !(other == *this); }
81
82 /// Exchange this buffer's values with the other buffer's values.
83 inline void swap(LeafBuffer&);
84
85 /// Return the memory footprint of this buffer in bytes.
86 inline Index memUsage() const;
87 /// Return the number of values contained in this buffer.
88 static Index size() { return SIZE; }
89
90 /// @brief Return a const pointer to the array of voxel values.
91 /// @warning The buffer must already be allocated (call @c allocate() first).
92 /// First-touch allocation via @c data() is not thread-safe; concurrent access
93 /// to an unallocated buffer is a programming error flagged in debug builds.
94 /// @warning This method should only be used by experts seeking low-level optimizations.
95 const ValueType* data() const;
96 /// @brief Return a pointer to the array of voxel values.
97 /// @warning The buffer must already be allocated (call @c allocate() first).
98 /// First-touch allocation via @c data() is not thread-safe; concurrent access
99 /// to an unallocated buffer is a programming error flagged in debug builds.
100 /// @warning This method should only be used by experts seeking low-level optimizations.
101 ValueType* data();
102
103private:
104 /// If this buffer is empty, return zero, otherwise return the value at index @ i.
105 inline const ValueType& at(Index i) const;
106
107 /// @brief Return a non-const reference to the value at index @a i.
108 /// @details This method is private since it makes assumptions about the
109 /// buffer's memory layout. LeafBuffers associated with custom leaf node types
110 /// (e.g., a bool buffer implemented as a bitmask) might not be able to
111 /// return non-const references to their values.
112 ValueType& operator[](Index i) { return const_cast<ValueType&>(this->at(i)); }
113
114 bool deallocate();
115
116 ValueType* mData = nullptr;
117#if OPENVDB_ABI_VERSION_NUMBER < 14
118 // Deprecated members kept for ABI compatibility
119 std::atomic<Index32> mDeprecatedAtomic{0};
120 tbb::spin_mutex mDeprecatedSpinMutex;
121#endif
122
123 friend class ::TestLeaf;
124 // Allow the parent LeafNode to access this buffer's data pointer.
125 template<typename, Index> friend class LeafNode;
126}; // class LeafBuffer

133inline
134LeafBuffer<T, Log2Dim>::LeafBuffer(const ValueType& val)

149template<typename T, Index Log2Dim>
150inline
151LeafBuffer<T, Log2Dim>::LeafBuffer(const LeafBuffer& other)
152 : mData(nullptr)
153{
154 if (other.mData != nullptr) {
155 this->allocate();
156 ValueType* target = mData;
157 const ValueType* source = other.mData;
158 Index n = SIZE;
159 while (n--) *target++ = *source++;
160 }
161}

164template<typename T, Index Log2Dim>
165inline void
166LeafBuffer<T, Log2Dim>::setValue(Index i, const ValueType& val)
167{
168 OPENVDB_ASSERT(i < SIZE);
169 if (mData) mData[i] = val;
170}

173template<typename T, Index Log2Dim>
174inline LeafBuffer<T, Log2Dim>&
175LeafBuffer<T, Log2Dim>::operator=(const LeafBuffer& other)
176{
177 if (&other != this) {
178 if (other.mData != nullptr) {
179 this->allocate();
180 ValueType* target = mData;
181 const ValueType* source = other.mData;
182 Index n = SIZE;
183 while (n--) *target++ = *source++;
184 }
185 }
186 return *this;
187}

190template<typename T, Index Log2Dim>
191inline void
192LeafBuffer<T, Log2Dim>::fill(const ValueType& val)
193{
194 if (mData != nullptr) {
195 ValueType* target = mData;
196 Index n = SIZE;
197 while (n--) *target++ = val;
198 }
199}

202template<typename T, Index Log2Dim>
203inline bool
204LeafBuffer<T, Log2Dim>::operator==(const LeafBuffer& other) const
205{
206 const ValueType *target = mData, *source = other.mData;
207 if (!target && !source) return true;
208 if (!target || !source) return false;
209 Index n = SIZE;
210 while (n && math::isExactlyEqual(*target++, *source++)) --n;
211 return n == 0;
212}

225LeafBuffer<T, Log2Dim>::memUsage() const
226{

233template<typename T, Index Log2Dim>
234inline const typename LeafBuffer<T, Log2Dim>::ValueType*
235LeafBuffer<T, Log2Dim>::data() const
236{
237 OPENVDB_ASSERT(mData != nullptr);
238 if (mData == nullptr) {
239 LeafBuffer* self = const_cast<LeafBuffer*>(this);
240 self->mData = new ValueType[SIZE];
241 }
242 return mData;
243}

246inline typename LeafBuffer<T, Log2Dim>::ValueType*
247LeafBuffer<T, Log2Dim>::data()
248{
249 OPENVDB_ASSERT(mData != nullptr);
250 if (mData == nullptr) mData = new ValueType[SIZE];
251 return mData;
252}

255template<typename T, Index Log2Dim>
256inline const typename LeafBuffer<T, Log2Dim>::ValueType&
257LeafBuffer<T, Log2Dim>::at(Index i) const
258{
259 static const ValueType sZero = zeroVal<T>();
260 OPENVDB_ASSERT(i < SIZE);
261 // We can't use the ternary operator here, otherwise Visual C++ returns
262 // a reference to a temporary.
263 if (mData) return mData[i]; else return sZero;
264}
265
266
267template<typename T, Index Log2Dim>
268inline bool
269LeafBuffer<T, Log2Dim>::deallocate()
270{
271 if (mData != nullptr) {
272 delete[] mData;
273 mData = nullptr;
274 return true;
275 }
276 return false;
277}

283// Partial specialization for bool ValueType
284template<Index Log2Dim>
285class LeafBuffer<bool, Log2Dim>
286{
287public:
288 using NodeMaskType = util::NodeMask<Log2Dim>;
289 using WordType = typename NodeMaskType::Word;
290 using ValueType = bool;
291 using StorageType = WordType;
292
293 static const Index WORD_COUNT = NodeMaskType::WORD_COUNT;
294 static const Index SIZE = 1 << 3 * Log2Dim;
295
296 static inline const bool sOn = true;
297 static inline const bool sOff = false;
298
299 LeafBuffer() {}
300 LeafBuffer(bool on): mData(on) {}
301 LeafBuffer(const NodeMaskType& other): mData(other) {}
302 LeafBuffer(const LeafBuffer& other): mData(other.mData) {}
303 ~LeafBuffer() {}
304 void fill(bool val) { mData.set(val); }
305 LeafBuffer& operator=(const LeafBuffer& b) { if (&b != this) { mData=b.mData; } return *this; }
306
307 const bool& getValue(Index i) const
308 {
309 OPENVDB_ASSERT(i < SIZE);
310 // We can't use the ternary operator here, otherwise Visual C++ returns
311 // a reference to a temporary.
312 if (mData.isOn(i)) return sOn; else return sOff;
313 }
314 const bool& operator[](Index i) const { return this->getValue(i); }
315
316 bool operator==(const LeafBuffer& other) const { return mData == other.mData; }
317 bool operator!=(const LeafBuffer& other) const { return mData != other.mData; }
318
319 void setValue(Index i, bool val) { OPENVDB_ASSERT(i < SIZE); mData.set(i, val); }
320
321 void swap(LeafBuffer& other) { if (&other != this) std::swap(mData, other.mData); }
322
323 Index memUsage() const { return sizeof(*this); }
324 static Index size() { return SIZE; }
325
326 /// @brief Return a pointer to the C-style array of words encoding the bits.
327 /// @warning This method should only be used by experts seeking low-level optimizations.
328 WordType* data() { return &(mData.template getWord<WordType>(0)); }
329 /// @brief Return a const pointer to the C-style array of words encoding the bits.
330 /// @warning This method should only be used by experts seeking low-level optimizations.
331 const WordType* data() const { return const_cast<LeafBuffer*>(this)->data(); }
332
333 /// @brief Return raw LeafBuffer data
334 const NodeMaskType& storage() const { return mData; }
335
336private:
337 // Allow the parent LeafNode to access this buffer's data.
338 template<typename, Index> friend class LeafNode;
339
340 NodeMaskType mData;
341}; // class LeafBuffer
