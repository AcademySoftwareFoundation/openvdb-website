39/// @brief Receives formatted log messages. Subclass this to route OpenVDB
40/// diagnostics into a host application.
41/// @details Sinks are called on the thread that logged the message, so an
42/// implementation must be thread safe. A sink must not throw.
43///
44/// @par Example
45/// @code
46/// class MySink: public openvdb::logging::Sink
47/// {
48/// public:
49/// MySink(): Sink("mysink") {}
50/// void append(openvdb::logging::Level level, const std::string& message,
51/// const char* file, int line) override
52/// {
53/// // forward to the host application's own logging system
54/// }
55/// };
56///
57/// openvdb::logging::addSink(std::make_shared<MySink>());
58/// @endcode
59class OPENVDB_API Sink
60{
61public:
62 using Ptr = std::shared_ptr<Sink>;
63
64 /// @param name Unique identifier, used by removeSink() and findSink().
65 /// @param threshold Messages below this level are not delivered to this sink,
66 /// in addition to the global threshold from setLevel().
67 explicit Sink(const std::string& name, Level threshold = Level::Debug);
68 virtual ~Sink();
69
70 const std::string& name() const;
71
72 Level threshold() const;
73 void setThreshold(Level);
74
75 /// @param level Severity of the message.
76 /// @param message The formatted message text, without a trailing newline.
77 /// @param file Source file the message was logged from, may be null.
78 /// @param line Line in @a file, or 0 if unknown.
79 virtual void append(Level level, const std::string& message,
80 const char* file, int line) = 0;
81
82private:
83 const std::string mName;
84 std::atomic<Level> mThreshold;
85};

150#define OPENVDB_LOG(level, message) \
151 do { \
152 if (openvdb::logging::isEnabledFor(openvdb::logging::Level::level)) { \
153 std::ostringstream _buf; \
154 _buf << message; \
155 openvdb::logging::internal::dispatch( \
156 openvdb::logging::Level::level, _buf.str(), __FILE__, __LINE__); \
157 } \
158 } while (0)
159
160/// Log an info message of the form '<TT>someVar << "some text" << ...</TT>'.
161#define OPENVDB_LOG_INFO(message) OPENVDB_LOG(Info, message)
162/// Log a warning message of the form '<TT>someVar << "some text" << ...</TT>'.
163#define OPENVDB_LOG_WARN(message) OPENVDB_LOG(Warn, message)
164/// Log an error message of the form '<TT>someVar << "some text" << ...</TT>'.
165#define OPENVDB_LOG_ERROR(message) OPENVDB_LOG(Error, message)
166/// Log a fatal error message of the form '<TT>someVar << "some text" << ...</TT>'.
167#define OPENVDB_LOG_FATAL(message) OPENVDB_LOG(Fatal, message)
168#ifdef NDEBUG
169/// In debug builds only, log a debugging message of the form '<TT>someVar << "text" << ...</TT>'.
170#define OPENVDB_LOG_DEBUG(message)
171#else
172/// In debug builds only, log a debugging message of the form '<TT>someVar << "text" << ...</TT>'.
173#define OPENVDB_LOG_DEBUG(message) OPENVDB_LOG(Debug, message)
174#endif

176/// @warning Don't use this in performance-critical code.
